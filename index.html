const SHEET_NAME = "Leads";
const NOTIFY_EMAIL = "xuandinh.avg@gmail.com";
const LEAD_HEADERS = [
  "Submitted At",
  "Vai trò",
  "Loại form",
  "Kênh",
  "Name",
  "Phone",
  "Email",
  "Company",
  "Facility Type",

function getTargetSpreadsheet() {
  const spreadsheetId = PropertiesService.getScriptProperties().getProperty("SPREADSHEET_ID");
  if (spreadsheetId) {
    return SpreadsheetApp.openById(spreadsheetId);
  }

  const activeSpreadsheet = SpreadsheetApp.getActiveSpreadsheet();
  if (activeSpreadsheet) {
    return activeSpreadsheet;
  }

  // Kết nối mặc định đến trang tính
  return SpreadsheetApp.openById("100vzbwgIwaJrqtAOaknwMxxILTMiGEhuVt8QX7J2Dpo");
}

function ensureLeadHeaders(sheet) {
  const existingHeaders = sheet.getLastRow() > 0 ? 
    sheet.getRange(1, 1, 1, Math.max(sheet.getLastColumn(), LEAD_HEADERS.length)).getValues()[0] : [];
  
  const hasRoleColumns =
    existingHeaders[1] === "Vai trò" &&
    existingHeaders[2] === "Loại form" &&
    existingHeaders[3] === "Kênh";

  if (hasRoleColumns) {
    return;
  }

  if (sheet.getLastRow() > 0 && sheet.getLastColumn() >= 1) {
    sheet.insertColumnsAfter(1, 3);
  }

  sheet.getRange(1, 1, 1, LEAD_HEADERS.length).setValues([LEAD_HEADERS]);
}

function resetLeadsSheet() {
  const ss = getTargetSpreadsheet();
  const sheet = ss.getSheetByName(SHEET_NAME) || ss.insertSheet(SHEET_NAME);
  sheet.clear();

  if (sheet.getMaxColumns() < LEAD_HEADERS.length) {
    sheet.insertColumnsAfter(sheet.getMaxColumns(), LEAD_HEADERS.length - sheet.getMaxColumns());
  }

  sheet.getRange(1, 1, 1, LEAD_HEADERS.length).setValues([LEAD_HEADERS]);
  sheet.setFrozenRows(1);
  return "Reset sheet " + SHEET_NAME + " with " + LEAD_HEADERS.length + " headers.";
}

// --- ĐỌC DỮ LIỆU TỪ SHEET (doGet) ---
function doGet(e) {
  try {
    var ss = getTargetSpreadsheet();
    var sheet = ss.getSheetByName(SHEET_NAME) || ss.insertSheet(SHEET_NAME);
    ensureLeadHeaders(sheet);
    
    var rows = sheet.getDataRange().getValues();
    var data = [];
    var headers = rows[0].map(function(h) { return h.toString().trim(); });
    
    for (var i = 1; i < rows.length; i++) {
      var row = rows[i];
      var obj = {};
      for (var j = 0; j < headers.length; j++) {
        obj[headers[j]] = row[j];
      }
      data.push(obj);
    }
    
    return ContentService.createTextOutput(JSON.stringify(data))
      .setMimeType(ContentService.MimeType.JSON)
      .setHeader('Access-Control-Allow-Origin', '*');
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.toString() }))
      .setMimeType(ContentService.MimeType.JSON)
      .setHeader('Access-Control-Allow-Origin', '*');
  }
}

// --- GHI / XÓA DỮ LIỆU THỜI GIAN THỰC (doPost) ---
function doPost(e) {
  var result = { status: "error", message: "Lỗi không xác định" };
  try {
    var jsonString = e.postData.contents;
    var payload = JSON.parse(jsonString);
    
    var ss = getTargetSpreadsheet();
    var sheet = ss.getSheetByName(SHEET_NAME) || ss.insertSheet(SHEET_NAME);
    ensureLeadHeaders(sheet);
    
    var rows = sheet.getDataRange().getValues();
    var headers = rows[0].map(function(h) { return h.toString().trim().toLowerCase(); });
    
    var colIndexes = {
      submittedAt: headers.indexOf("submitted at"),
      role: headers.indexOf("vai trò"),
      formType: headers.indexOf("loại form"),
      channel: headers.indexOf("kênh"),
      name: headers.indexOf("name") !== -1 ? headers.indexOf("name") : headers.indexOf("họ và tên"),
      phone: headers.indexOf("phone") !== -1 ? headers.indexOf("phone") : headers.indexOf("số điện thoại"),
      email: headers.indexOf("email"),
      company: headers.indexOf("company") !== -1 ? headers.indexOf("company") : headers.indexOf("công ty"),
      facilityType: headers.indexOf("facility type"),
      interestedIn: headers.indexOf("interested in"),
      purchaseScale: headers.indexOf("purchase scale"),
      status: headers.indexOf("trạng thái"),
      deliveryFrequency: headers.indexOf("delivery frequency"),
      deliveryArea: headers.indexOf("delivery area"),
      needBy: headers.indexOf("need by"),
      message: headers.indexOf("message") !== -1 ? headers.indexOf("message") : headers.indexOf("ghi chú"),
      selectedItems: headers.indexOf("selected items"),
      selectedCount: headers.indexOf("selected count"),
      source: headers.indexOf("source") !== -1 ? headers.indexOf("source") : headers.indexOf("nguồn")
    };

    // ĐỒNG BỘ TỪ DASHBOARD (Có chứa action)
    if (payload.action) {
      var action = payload.action;
      var data = payload.data;

      if (action === "add") {
        var newRow = [];
        var maxCols = Math.max(sheet.getLastColumn(), LEAD_HEADERS.length);
        for (var k = 0; k < maxCols; k++) { newRow.push(""); }
        
        newRow[colIndexes.submittedAt] = new Date().toISOString();
        newRow[colIndexes.role] = "Người mua";
        newRow[colIndexes.formType] = "Tạo thủ công";
        newRow[colIndexes.channel] = "Dashboard";
        newRow[colIndexes.name] = data.name || "";
        newRow[colIndexes.phone] = data.phone || "";
        newRow[colIndexes.email] = data.email || "";
        newRow[colIndexes.company] = data.company || "";
        
        var catText = "Khách lẻ - Thường";
        if (data.category === "wholesale_restaurant") catText = "Khách sỉ - Nhà hàng";
        else if (data.category === "wholesale_agency") catText = "Khách sỉ - Đại lý";
        else if (data.category === "retail_vip") catText = "Khách lẻ - VIP";
        
        // Trạng thái tiếng Việt
        var statusLabel = "Mới";
        if (data.status === "contacting") statusLabel = "Đã liên hệ";
        else if (data.status === "quoting") statusLabel = "Đang báo giá";
        else if (data.status === "quoted") statusLabel = "Đã báo giá";
        else if (data.status === "won") statusLabel = "Đã chốt đơn";
        else if (data.status === "unqualified") statusLabel = "Không tiềm năng";
        else if (data.status === "canceled") statusLabel = "Hủy";
        newRow[colIndexes.status] = statusLabel;
        
        newRow[colIndexes.message] = "Phân loại: " + catText;
        newRow[colIndexes.source] = data.source || "Website";

        sheet.appendRow(newRow);
        result = { status: "success", message: "Đã thêm lead từ Dashboard thành công!" };
        
      } else if (action === "update_status") {
        var targetPhone = data.phone.toString().replace(/[^0-9+]/g, '');
        var newStatus = data.status;
        
        var statusLabel = "Mới";
        if (newStatus === "contacting") statusLabel = "Đã liên hệ";
        else if (newStatus === "quoting") statusLabel = "Đang báo giá";
        else if (newStatus === "quoted") statusLabel = "Đã báo giá";
        else if (newStatus === "won") statusLabel = "Đã chốt đơn";
        else if (newStatus === "unqualified") statusLabel = "Không tiềm năng";
        else if (newStatus === "canceled") statusLabel = "Hủy";
        
        var updatedCount = 0;
        for (var i = 1; i < rows.length; i++) {
          var rowPhone = rows[i][colIndexes.phone].toString().replace(/[^0-9+]/g, '');
          if (rowPhone === targetPhone) {
            sheet.getRange(i + 1, colIndexes.status + 1).setValue(statusLabel);
            updatedCount++;
          }
        }
        result = { status: "success", message: "Đã cập nhật trạng thái cho " + updatedCount + " dòng thành " + statusLabel };
        
      } else if (action === "delete") {
        var targetPhone = data.phone.toString().replace(/[^0-9+]/g, '');
        var deletedCount = 0;
        
        // Quét ngược từ dưới lên để tránh lệch hàng khi xóa
        for (var i = rows.length - 1; i >= 1; i--) {
          var rowPhone = rows[i][colIndexes.phone].toString().replace(/[^0-9+]/g, '');
          if (rowPhone === targetPhone) {
            sheet.deleteRow(i + 1);
            deletedCount++;
          }
        }
        result = { status: "success", message: "Đã xóa " + deletedCount + " dòng!" };
      }
      
    } else {
      // NHẬN TỪ WEBSITE FORM (Không chứa action, chứa payload trực tiếp)
      var newRow = [];
      var maxCols = Math.max(sheet.getLastColumn(), LEAD_HEADERS.length);
      for (var k = 0; k < maxCols; k++) { newRow.push(""); }
      
      var submittedTime = new Date();
      var formattedDate = Utilities.formatDate(submittedTime, "GMT+7", "dd/MM/yyyy HH:mm:ss");
      
      newRow[colIndexes.submittedAt] = submittedTime.toISOString();
      newRow[colIndexes.role] = payload.role || "";
      newRow[colIndexes.formType] = payload.formType || "";
      newRow[colIndexes.channel] = payload.channel || "";
      newRow[colIndexes.name] = payload.name || "";
      newRow[colIndexes.phone] = payload.phone || "";
      newRow[colIndexes.email] = payload.email || "";
      newRow[colIndexes.company] = payload.company || "";
      newRow[colIndexes.facilityType] = payload.facilityType || "";
      newRow[colIndexes.interestedIn] = payload.interestedIn || "";
      newRow[colIndexes.purchaseScale] = payload.purchaseScale || "";
      newRow[colIndexes.status] = "Mới"; // Mặc định trạng thái Mới
      newRow[colIndexes.deliveryFrequency] = payload.deliveryFrequency || "";
      newRow[colIndexes.deliveryArea] = payload.deliveryArea || "";
      newRow[colIndexes.needBy] = payload.needBy || "";
      newRow[colIndexes.message] = payload.message || "";
      
      var selectedItems = "";
      if (payload.cart && Array.isArray(payload.cart)) {
        selectedItems = payload.cart.map(function(item) {
          return (item.name || item.title) + " (SL: " + (item.quantity || item.qty) + ")";
        }).join(", ");
      } else if (payload.selectedItems) {
        selectedItems = payload.selectedItems;
      }
      newRow[colIndexes.selectedItems] = selectedItems;
      newRow[colIndexes.selectedCount] = payload.selectedCount || (payload.cart ? payload.cart.length : "");
      newRow[colIndexes.source] = payload.source || "";
      newRow[colIndexes.rawPayload] = JSON.stringify(payload);
      
      sheet.appendRow(newRow);
      var insertedRowIndex = sheet.getLastRow();
      
      // GỬI EMAIL THÔNG BÁO CHO ANH ĐỊNH
      try {
        var emailBody = "
    CÓ KHÁCH HÀNG LIÊN HỆ MỚI (MẪU BÁO GIÁ)
    " +
          "
    Thời gian: " + formattedDate + "
    " +
          "
    Họ và tên: " + (payload.name || "") + "
    " +
          "
    Số điện thoại: " + (payload.phone || "") + "
    " +
          "
    Email: " + (payload.email || "") + "
    " +
          "
    Vai trò: " + (payload.role || "") + "
    " +
          "
    Kênh liên hệ: " + (payload.channel || "") + "
    " +
          "
    Công ty: " + (payload.company || "") + "
    " +
          "
    Loại hình đơn vị: " + (payload.facilityType || "") + "
    " +
          "
    Mặt hàng quan tâm: " + (payload.interestedIn || "") + "
    " +
          "
    Quy mô nhu cầu: " + (payload.purchaseScale || "") + "
    " +
          "
    Tần suất giao: " + (payload.deliveryFrequency || "") + "
    " +
          "
    Khu vực giao: " + (payload.deliveryArea || "") + "
    " +
          "
    Cần trước: " + (payload.needBy || "") + "
    " +
          "
    Sản phẩm đã chọn trên web: " + (selectedItems || "Không có") + "
    " +
          "
    Lời nhắn:
    " + String(payload.message || "").replace(/\n/g, "
    ") + "
    " +
          "
    " +
          "
    Dữ liệu đã tự động đồng bộ lên Google Sheet dòng: " + insertedRowIndex + "
    ";
          
        MailApp.sendEmail({
          to: NOTIFY_EMAIL,
          subject: "🔥 LEAD BÁO GIÁ MỚI: " + (payload.name || "") + " (" + (payload.phone || "") + ")",

      } catch (emailErr) {
        console.error("Lỗi gửi email thông báo: " + emailErr.toString());
      }
      
      result = { status: "success", message: "Đã ghi nhận form báo giá website thành công!" };

  } catch (error) {

  }

  return ContentService.createTextOutput(JSON.stringify(result))

}
